//! T-171 scratch: a day on the real New York network, commutes and (optionally) the everyday
//! layer, with route-node boardings per purpose, transfers touching a chosen feed, and trip
//! explorer answers for a list of trips. Scratch copy only, not in the shared tree.
//!
//!   t171net --pack P --net N --out O [--rounds 3] [--everyday] [--fares] [--feed path]
//!           [--trips trips.json] [--set k=v,..]

use serde_json::{json, Value};
use trainworld_sim::demand::api::DemandApi;
use trainworld_sim::demand::everyday::EverydayApi;
use trainworld_sim::demand::network::PERIODS;
use trainworld_sim::demand::pack;

fn arr_f32(v: &Value, k: &str) -> Vec<f32> {
    v[k].as_array().unwrap().iter().map(|x| x.as_f64().unwrap() as f32).collect()
}
fn arr_u32(v: &Value, k: &str) -> Vec<u32> {
    v[k].as_array().unwrap().iter().map(|x| x.as_u64().unwrap() as u32).collect()
}
fn r1(v: &[f32]) -> Vec<f64> {
    v.iter().map(|&x| (x as f64 * 10.0).round() / 10.0).collect()
}

fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    let get = |k: &str| args.iter().position(|a| a == k).map(|i| args[i + 1].clone());
    let has = |k: &str| args.iter().any(|a| a == k);
    let pack_path = get("--pack").expect("--pack");
    let net_path = get("--net").expect("--net");
    let out_path = get("--out").expect("--out");
    let rounds: usize = get("--rounds").map_or(3, |v| v.parse().unwrap());
    let feed = get("--feed").unwrap_or("path".into());
    let (h, b) = pack::load_raw(&pack_path).unwrap();
    let mut api = DemandApi::new(&h, &b).unwrap();
    let mut ev = if has("--everyday") { Some(EverydayApi::new(&h, &b).unwrap()) } else { None };
    if let Some(v) = get("--set") {
        for kv in v.split(',') {
            let (k, x) = kv.split_once('=').unwrap();
            let x: f32 = x.parse().unwrap();
            let p = api.params_mut();
            match k {
                "w_drive" => p.w_drive = x,
                "park_min" => p.park_min = x,
                "fare_min" => p.fare_min = x,
                "xfer_walk_m" => p.xfer_walk_m = x,
                _ => p.set_extra(k, x).unwrap_or_else(|e| panic!("{e}")),
            }
        }
    }
    eprintln!("commute info {:?}", api.info());
    if let Some(e) = &ev {
        eprintln!("everyday info {:?}", e.info());
    }
    let net: Value = serde_json::from_str(&std::fs::read_to_string(&net_path).unwrap()).unwrap();
    let d = &net["demand"];
    let (st_xy, line_n, stops, times, tph, cars) = (arr_f32(d, "st_xy"), arr_u32(d, "line_n"), arr_u32(d, "stops"), arr_f32(d, "times"), arr_f32(d, "tph"), arr_f32(d, "cars"));
    api.set_network(&st_xy, &line_n, &stops, &times, &tph, &cars).unwrap();
    if let Some(e) = ev.as_mut() {
        // the everyday engine has its own DemandApi inside with default params
        e.set_network(&st_xy, &line_n, &stops, &times, &tph, &cars).unwrap();
    }
    // route node -> line, as the kernel lays them out: each line's forward then reverse stops
    let mut hop_km = vec![];
    let mut rn_line_net = vec![];
    let mut off = 0usize;
    for (l, &n) in line_n.iter().enumerate() {
        let n = n as usize;
        for dir in 0..2 {
            for i in 0..n {
                let k = if dir == 0 { off + i } else { off + n - 1 - i };
                rn_line_net.push(l);
                hop_km.push(if i + 1 < n {
                    let nx = if dir == 0 { k + 1 } else { k - 1 };
                    let (s0, s1) = (stops[k] as usize, stops[nx] as usize);
                    (st_xy[2 * s0] - st_xy[2 * s1]).hypot(st_xy[2 * s0 + 1] - st_xy[2 * s1 + 1]) / 1000.0
                } else {
                    0.0
                });
            }
        }
        off += n;
    }
    if has("--fares") {
        api.set_fares(2.9, 0.1, &hop_km).unwrap();
        if let Some(e) = ev.as_mut() {
            e.set_fares(2.9, 0.1, &hop_km).unwrap();
        }
    }
    let lines = net["lines"].as_array().unwrap();
    let sel: Vec<u8> = rn_line_net.iter().map(|&l| (feed == "all" || lines[l]["feed"].as_str().unwrap() == feed) as u8).collect();
    let counties: Option<Vec<u8>> = get("--counties").map(|p| std::fs::read(p).unwrap());
    let mut periods = vec![];
    for q in 0..PERIODS {
        api.solve(q);
        if let Some(e) = ev.as_mut() {
            e.solve(q);
        }
        for _ in 0..rounds {
            let mut total = api.seg(q);
            if let Some(e) = &ev {
                for (t, x) in total.iter_mut().zip(e.seg(q)) {
                    *t += x;
                }
            }
            api.crowd_with_external(q, &total);
            if let Some(e) = ev.as_mut() {
                e.crowd_with_external(q, &total);
            }
        }
        let s = api.summary(q);
        let mut o = json!({
            "summary": s,
            "board": r1(&api.board(q)),
            "alight": r1(&api.alight(q)),
            "seg": r1(&api.seg(q)),
            "entries": r1(&api.entries(q)),
            "exits": r1(&api.exits(q)),
        });
        if let Some(e) = &ev {
            o["ev_summary"] = json!(e.summary(q));
            o["ev_kinds"] = json!(e.kind_totals(q));
            o["ev_parts"] = json!(r1(&e.part_loads(q)));
        }
        if has("--cells") {
            o["rail_cell"] = json!(r1(&api.rail_by_cell(q)));
            o["trips_cell"] = json!(r1(&api.trips_by_cell(q)));
        }
        if let Some(c) = &counties {
            let n = 25usize;
            let rail = api.rail_by_cell(q);
            let trips = api.trips_by_cell(q);
            let (mut r, mut t) = (vec![0f64; n], vec![0f64; n]);
            for (i, &k) in c.iter().enumerate() {
                if (k as usize) < n && i < rail.len() {
                    r[k as usize] += rail[i] as f64;
                    t[k as usize] += trips[i] as f64;
                }
            }
            o["county"] = json!({"rail": r, "trips": t});
        }
        eprintln!("period {q}: commute rail {:.0} of {:.0}", s[2], s[1]);
        periods.push(o);
    }
    let xfer = api.transfers(&sel);
    let ev_xfer = ev.as_ref().map(|e| e.transfers(&sel)).unwrap_or_default();
    let (rn_station, rn_line) = api.route_nodes(0);
    // trips: [{name, from: [x, y], to: [x, y]}] in the pack frame, morning peak and midday
    let mut trips_out = vec![];
    if let Some(tp) = get("--trips") {
        let tj: Value = serde_json::from_str(&std::fs::read_to_string(tp).unwrap()).unwrap();
        for t in tj.as_array().unwrap() {
            let f = t["from"].as_array().unwrap();
            let to = t["to"].as_array().unwrap();
            let (ox, oy, dx, dy) = (f[0].as_f64().unwrap() as f32, f[1].as_f64().unwrap() as f32, to[0].as_f64().unwrap() as f32, to[1].as_f64().unwrap() as f32);
            trips_out.push(json!({"name": t["name"], "am": api.trip(0, ox, oy, dx, dy), "midday": api.trip(1, ox, oy, dx, dy)}));
        }
    }
    let out = json!({
        "net": net_path,
        "everyday": ev.is_some(),
        "ev_info": ev.as_ref().map(|e| e.info()),
        "commute_info": api.info(),
        "rn_station": rn_station,
        "rn_line": rn_line,
        "st_map": api.station_map(),
        "periods": periods,
        "xfer": r1(&xfer),
        "ev_xfer": r1(&ev_xfer),
        "trips": trips_out,
    });
    std::fs::write(&out_path, serde_json::to_string(&out).unwrap()).unwrap();
    eprintln!("-> {out_path}");
}
