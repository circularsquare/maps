### phase2.sh

```bash
#!/usr/bin/env bash
# T-4159 phase 2, run as root on the box: move trainworld games onto the volume.
# Start it detached from ssh: systemd-run --unit=t4159-phase2 --collect bash ${0}; read: journalctl -u t4159-phase2 -f
# Any failure after the stop runs rollback.sh, which puts everything back and starts shonei.
set -uo pipefail
V=/mnt/HC_Volume_107598115
SRC=/opt/shonei/trainworld/games
DST=$V/trainworld-games
OLD=/root/tw-games-pre-volume
ST=$V/volmove
BIN=/opt/shonei/shonei-market-linux
say() { echo "[$(date -u +%T)] $*"; }

FSTAB_LINE="$DST $SRC none bind,nofail,x-systemd.requires-mounts-for=$V,x-systemd.required-by=shonei.service,x-systemd.before=shonei.service 0 0"
die()  { say "FAILED before stop: $*"; exit 1; }
back() { say "FAILED: $* -- rolling back"; bash "$ST/rollback.sh"; exit 1; }

# --- preflight, service still up -------------------------------------------
findmnt -n "$V" >/dev/null || die "volume not mounted"
findmnt -n "$SRC" >/dev/null && die "something already mounted at $SRC"
[ -e "$OLD" ] && die "$OLD already exists"
[ -x "$ST/shonei-market-linux.new" ] && [ -f "$ST/shonei-market-linux.prev" ] || die "staged binaries missing"
systemctl is-active --quiet shonei-backup && die "backup is running, wait for it"
grep -q "trainworld-games" /etc/fstab && die "fstab already has the line"
say "warm rsync"
rsync -a --numeric-ids --delete "$SRC/" "$DST/" || die "warm rsync"
cp -p /etc/fstab /root/fstab.pre-T-4159 || die "fstab backup"

# --- downtime starts -----------------------------------------------------------
T0=$(date +%s)
say "stop shonei"
systemctl stop shonei
systemctl is-active --quiet shonei && back "shonei still active after stop"
say "final rsync"
rsync -a --numeric-ids --delete "$SRC/" "$DST/" || back "final rsync"
n=$(rsync -an --numeric-ids --delete --itemize-changes "$SRC/" "$DST/" | wc -l)
[ "$n" -eq 0 ] || back "copy differs after final rsync ($n lines)"
say "switch"
mv "$SRC" "$OLD" || back "mv old dir"
install -d -o root -g root -m 000 "$SRC" || back "mkdir mountpoint"   # unwritable if ever not mounted
echo "$FSTAB_LINE" >> /etc/fstab || back "fstab append"
systemctl daemon-reload
systemctl start opt-shonei-trainworld-games.mount || back "mount unit"
findmnt -n -o SOURCE "$SRC" | grep -q "trainworld-games" || back "bind mount not in place"
a=$(ls "$SRC" | wc -l); b=$(ls "$OLD" | wc -l)
[ "$a" -eq "$b" ] && [ "$a" -gt 9000 ] || back "account folders $a on volume, $b old"
systemctl show shonei -p Requires | tr ' ' '\n' | grep -q opt-shonei-trainworld-games.mount || back "shonei does not require the mount"
say "binary with the raised cap"
install -m 755 -o shonei -g shonei "$ST/shonei-market-linux.new" "$BIN" || back "install binary"
say "start shonei"
systemctl start shonei
up=""
for i in $(seq 1 60); do
  if journalctl -u shonei --since "@$T0" --no-pager | grep -q "server starting"; then up=1; break; fi
  sleep 0.5
done
T1=$(date +%s)
[ -n "$up" ] && systemctl is-active --quiet shonei || back "shonei did not come up"
say "DOWNTIME $((T1-T0)) s"

# --- checks, service up ------------------------------------------------------
journalctl -u shonei --since "@$T0" --no-pager | grep -E '\[trainworld\] store at|server starting'
curl -s -m 5 http://127.0.0.1:8083/noritetsu/config | grep -o '"trainworldGames":[a-z]*'
say "waiting up to 180 s for a game save to land on the volume"
for i in $(seq 1 36); do
  f=$(find "$DST" -type f -newermt "@$T1" ! -name '*.tmp' | head -3)
  [ -n "$f" ] && break; sleep 5
done
[ -n "$f" ] && { say "new files on the volume:"; echo "$f" | sed "s#$DST/[^/]*/#<acct>/#"; } || say "WARN no new save in 180 s"
acct=$(ls "$SRC" | head -1)
say "backup tar sees the games through the mount: $(tar -C /opt/shonei -cf - "./trainworld/games/$acct" | tar -tf - | wc -l) entries for one account"
df -h "$V" /
say "done"
```

### rollback.sh

```bash
#!/usr/bin/env bash
# T-4159 rollback, run as root on the box: games back on the root disk at the old path, old binary, start.
# Copies anything saved on the volume since the switch back first, so it is safe after hours too.
set -uo pipefail
V=/mnt/HC_Volume_107598115
SRC=/opt/shonei/trainworld/games
DST=$V/trainworld-games
OLD=/root/tw-games-pre-volume
ST=$V/volmove
BIN=/opt/shonei/shonei-market-linux
say() { echo "[$(date -u +%T)] $*"; }

systemctl stop shonei
if findmnt -n "$SRC" >/dev/null; then
  if [ -d "$OLD" ]; then
    say "copy saves made on the volume back to $OLD"
    rsync -a --numeric-ids --delete "$DST/" "$OLD/" || { say "rsync back FAILED, stopping here (shonei stopped)"; exit 1; }
  fi
  umount "$SRC" || { say "umount FAILED, stopping here (shonei stopped)"; exit 1; }
fi
if [ -f /root/fstab.pre-T-4159 ]; then cp -p /root/fstab.pre-T-4159 /etc/fstab; fi
systemctl daemon-reload
if [ -d "$OLD" ]; then
  rmdir "$SRC" 2>/dev/null
  [ -e "$SRC" ] && { say "$SRC is not empty, not moving $OLD back (shonei stopped)"; exit 1; }
  mv "$OLD" "$SRC"
fi
[ -f "$ST/shonei-market-linux.prev" ] && install -m 755 -o shonei -g shonei "$ST/shonei-market-linux.prev" "$BIN"
systemctl start shonei
sleep 4
say "shonei: $(systemctl is-active shonei); games at $SRC: $(ls "$SRC" | wc -l) accounts; mount: $(findmnt -n -o SOURCE "$SRC" || echo none)"
```
