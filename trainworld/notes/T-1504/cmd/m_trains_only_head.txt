// T-1504 probe command: experiment on the every-pair page after the half fare: China's fare scale
// back to its income's, and six times the trains on the high-speed lines (no passages).
const w = window.tw.world.value;
await window.tw.client.edit({ op: 'fareScale', country: 'CN', scale: null });
const note = [];
for (const l of w.save.lines.filter((l) => Number(l.id) >= 113)) {
  const r = await window.tw.client.edit({ op: 'setTrainCount', line: Number(l.id), trains: [36, 18, 12] });
  note.push([l.id, r?.ok]);
}
